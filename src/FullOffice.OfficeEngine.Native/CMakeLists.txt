cmake_minimum_required(VERSION 3.25)
project(FullOfficeOfficeEngineHost LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This host currently targets Windows")
endif()
option(ENGINE_COMPILE_ONLY "Compile the bridge without machine-local runtime binaries" OFF)
if(NOT ENGINE_COMPILE_ONLY AND NOT DEFINED ENGINE_IMPORT_LIBRARY)
    message(FATAL_ERROR "Provide the verified native core import library")
endif()
if(ENGINE_COMPILE_ONLY)
    add_library(FullOfficeOfficeHost OBJECT OfficeEngineHost.cpp)
else()
    add_executable(FullOfficeOfficeHost OfficeEngineHost.cpp)
endif()
target_compile_features(FullOfficeOfficeHost PRIVATE cxx_std_17)
target_compile_definitions(FullOfficeOfficeHost PRIVATE WIN32 UNICODE _UNICODE NOMINMAX)
target_compile_options(FullOfficeOfficeHost PRIVATE /utf-8 /W4)
target_include_directories(FullOfficeOfficeHost SYSTEM PRIVATE third_party/desktop-sdk/ChromiumBasedEditors/lib/include)
if(NOT ENGINE_COMPILE_ONLY)
    target_link_libraries(FullOfficeOfficeHost PRIVATE "${ENGINE_IMPORT_LIBRARY}" user32 shell32)
    target_link_options(FullOfficeOfficeHost PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup)
endif()
