cmake_minimum_required(VERSION 3.25)
project(FullToolsDistribution VERSION 0.1.0 LANGUAGES C CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "This distribution helper targets Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(miniz STATIC third_party/miniz/miniz.c)
target_include_directories(miniz PUBLIC third_party/miniz)
add_library(distribution_core STATIC BundleExtract.cpp DefaultApplications.cpp UpdateManifest.cpp UpdateDownload.cpp
    transaction/core/path_utf8.cpp transaction/core/sha256.cpp
    transaction/install/install_transaction.cpp transaction/install/product_registration.cpp)
target_include_directories(distribution_core PUBLIC . transaction)
target_link_libraries(distribution_core PUBLIC miniz bcrypt crypt32 ole32 shell32 winhttp)
target_compile_definitions(distribution_core PUBLIC UNICODE _UNICODE NOMINMAX)
target_compile_options(distribution_core PRIVATE /W4 /WX /utf-8 /permissive-)
configure_file(Helper.rc.in Helper.rc @ONLY)
add_executable(FullOfficeSetup FullOfficeSetup.cpp "${CMAKE_CURRENT_BINARY_DIR}/Helper.rc")
target_link_libraries(FullOfficeSetup PRIVATE distribution_core)
target_compile_options(FullOfficeSetup PRIVATE /W4 /WX /utf-8 /permissive-)
target_link_options(FullOfficeSetup PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup /MANIFEST:NO)
add_executable(ExtractionProbe ExtractionProbe.cpp)
target_link_libraries(ExtractionProbe PRIVATE distribution_core)
target_compile_options(ExtractionProbe PRIVATE /W4 /WX /utf-8 /permissive-)
add_executable(UpdateProbe UpdateProbe.cpp)
target_link_libraries(UpdateProbe PRIVATE distribution_core)
target_compile_options(UpdateProbe PRIVATE /W4 /WX /utf-8 /permissive-)
add_executable(UpdateDownloadProbe UpdateDownloadProbe.cpp)
target_link_libraries(UpdateDownloadProbe PRIVATE distribution_core)
target_compile_options(UpdateDownloadProbe PRIVATE /W4 /WX /utf-8 /permissive-)
add_executable(FullToolsUpdater UpdateWindow.cpp "${CMAKE_CURRENT_BINARY_DIR}/Helper.rc")
target_link_libraries(FullToolsUpdater PRIVATE distribution_core comctl32)
target_compile_options(FullToolsUpdater PRIVATE /W4 /WX /utf-8 /permissive-)
target_link_options(FullToolsUpdater PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup /MANIFEST:NO)
if(DEFINED PAYLOAD_ZIP)
    set(PRODUCT_TITLE "FullOffice" CACHE STRING "Installer title")
    set(PRODUCT_ID "FullOffice" CACHE STRING "Product identifier")
    set(BACKEND_NAME "FullOfficeSetup.exe" CACHE STRING "Native setup backend")
    set(BACKEND_USES_EQUALS "false" CACHE STRING "Native backend uses equals arguments")
    set(PRODUCT_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../FullOffice.Windows/Assets/FlowOffice.ico" CACHE FILEPATH "Product icon")
    file(SHA256 "${PAYLOAD_ZIP}" PAYLOAD_SHA256)
    configure_file(BundleConfig.h.in BundleConfig.h @ONLY)
    configure_file(Bundle.rc.in Bundle.rc @ONLY)
    add_executable(FullToolsInstaller BundleLauncher.cpp "${CMAKE_CURRENT_BINARY_DIR}/Bundle.rc")
    target_include_directories(FullToolsInstaller PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(FullToolsInstaller PRIVATE distribution_core comctl32)
    target_compile_options(FullToolsInstaller PRIVATE /W4 /WX /utf-8 /permissive-)
    target_link_options(FullToolsInstaller PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup /MANIFEST:NO)
endif()
